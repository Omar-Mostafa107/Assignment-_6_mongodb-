
// 1. Create an explicit "books" collection with a validation rule
// requiring a non-empty "title" field.
db.createCollection("books", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["title"],
      properties: {
        title: {
          bsonType: "string",
          minLength: 1,
          description: "title must be a non-empty string and is required",
        },
      },
    },
  },
});

// 2. Create an implicit "authors" collection by inserting data directly.
db.authors.insertOne({
  name: "Author1",
  nationality: "British",
});

// 3. Create a capped collection named "logs" with a size limit of 1MB.
db.createCollection("logs", {
  capped: true,
  size: 1048576, // 1MB
});

// 4. Create an index on the books collection for the title field.
db.books.createIndex({ title: 1 });

// 5. Insert one document into the books collection.
db.books.insertOne({
  title: "Book1",
  author: "Ali",
  year: 1937,
  genres: ["Fantasy", "Adventure"],
});

// 6. Insert multiple documents into the books collection (at least three).
db.books.insertMany([
  {
    title: "Future",
    author: "George Orwell",
    year: 2020,
    genres: ["Science Fiction"],
  },
  {
    title: "To Kill a Mockingbird",
    author: "Harper Lee",
    year: 1960,
    genres: ["Classic", "Fiction"],
  },
  {
    title: "Brave New World",
    author: "Aldous Huxley",
    year: 2006,
    genres: ["Dystopian", "Science Fiction"],
  },
]);

// 7. Insert a new log into the logs collection.
// Replace the ObjectId below with the real _id of a book from your books collection.
db.logs.insertOne({
  book_id: ObjectId("64b5c2d8a123456ef8905"),
  action: "borrowed",
});

// 8. Update the book with title "Future", changing its year to 2022.
db.books.updateOne({ title: "Future" }, { $set: { year: 2022 } });

// 9. Find a book with title "Brave New World".
db.books.findOne({ title: "Brave New World" });

// 10. Find all books published between 1990 and 2010.
db.books.find({ year: { $gte: 1990, $lte: 2010 } });

// 11. Find books where the genres array includes "Science Fiction".
db.books.find({ genres: "Science Fiction" });

// 12. Skip the first two books, limit to the next three, sorted by year descending.
db.books.find().sort({ year: -1 }).skip(2).limit(3);

// 13. Find books where the year field is stored as an integer (BSON type int).
db.books.find({ year: { $type: "int" } });

// 14. Find all books where genres does not include "Horror" or "Science Fiction".
db.books.find({ genres: { $nin: ["Horror", "Science Fiction"] } });

// 15. Delete all books published before 2000.
db.books.deleteMany({ year: { $lt: 2000 } });

// 16. Aggregation: filter books published after 2000, sorted by year descending.
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $sort: { year: -1 } },
]);

// 17. Aggregation: books published after 2000, showing only title, author, year.
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $project: { _id: 0, title: 1, author: 1, year: 1 } },
]);

// 18. Aggregation: break the genres array into separate documents.
db.books.aggregate([
  { $unwind: "$genres" },
  { $project: { _id: 0, title: 1, genres: 1 } },
]);

// 19. Aggregation: join the books collection with the logs collection.
db.logs.aggregate([
  {
    $lookup: {
      from: "books",
      localField: "book_id",
      foreignField: "_id",
      as: "book_details",
    },
  },
  {
    $project: {
      _id: 0,
      action: 1,
      book_details: { _id: 0, title: 1, author: 1, year: 1 },
    },
  },
]);
